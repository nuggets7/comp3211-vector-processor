Sample Benchmark

Same but 4 marks fields of 8 bits each, allowing us to use VADD
There is also no checks for duplicate IDs for simplicity
Stream raw dataset into DMEM

Each IN input will be an 32 bit value stored in a 32 bit register.
It will go in order of all 20 Ids, all 20 M1s, and so on, in groups of 4

R1 = Write pointer
R2 = field counter
R3 = Vector 1 location
R4 = Vector 2 location
R5 = Vector 3 location
R6 = Vector 4 location
R7 = Vector 5 location
R8 = Total Vector
R9 = Current Vector

R10 = Read Input
R11 = Totals pointer
R12 = Totals pointer offset


R30 = Mark cap (100)
R29 = total count (25)
R28 = total students (20)
R27 = inputs per type (5)
R26 = Vector element 1 OR (F000)
R25 = Vector element 2 OR (0F00)
R24 = Vector element 3 OR (00F0)
R23 = Vector element 4 OR (000F)
R22 = Final total vectors (4)
R21 = DIV shift numbers
R20 = Result

ADDI    R27, R27, 5
ADDI    R28, R28, 20
ADDI    R29, R29, 25 (100 values / 4 values per 32 bits = 25)
ADDI    R30, R30, 100


Read_Loop:
    IN      R10
    STORE   R10, 0(R1)
    ADDI    R1, R1, 1
    ADDI    R2, R2, 1
    NOOP
    NOOP
    NOOP
    BLT     R2, R29, Read_Loop


ADD     R11, R1, R0
ADD     R1, R0, R0
ADD     R2, R0, R0

Proc_Loop:
    LOAD    R3, 5(R1)
    LOAD    R4, 10(R1)
    LOAD    R5, 15(R1)
    LOAD    R6, 20(R1)
    LOAD    R7, 25(R1)
    VADD    R8, R3, R4
    VADD    R8, R8, R5
    VADD    R8, R8, R6
    VADD    R8, R8, R7
    VCLIP   R8, 100
    ADD     R12, R11, R1
    STORE   R8, 0(R12)
    ADDI    R1, R1, 1
    NOOP
    NOOP
    NOOP
    BLT     R1, R27, Proc_Loop


ADD     R1, R0, R0
ADD     R8, R0, R10

Avg_Totals:
    ADD     R12, R11, R1
    LOAD    R8, 0(R12)
    VADD    R9, R9, R8
    ADDI    R1, R1, 1
    NOOP
    NOOP
    NOOP
    BLT     R1, R27, Avg_Totals

ADDI    R26, R0, FF000000 (hex)
ADDI    R25, R0, 00FF0000 (hex)
ADDI    R24, R0, 0000FF00 (hex)
ADDI    R23, R0, 000000FF (hex)

OR      R26, R26, R9
OR      R25, R25, R9
OR      R24, R24, R9
OR      R23, R23, R9

ADDI    R21, 01000000 (hex)
DIV     R26, R26, R21

ADD     R21, R0, R0
ADDI    R21, 00010000 (hex)
DIV     R25, R25, R21

ADD     R21, R0, R0
ADDI    R21, 00000100 (hex)
DIV     R24, R24, R21

ADD     R20, R23, R24
ADD     R20, R20, R25
ADD     R20, R20, R26
ADDI    R22, R0, 4

DIV     R20, R20, R22
OUT     R20
